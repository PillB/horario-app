// ─── Internal Helpers ─────────────────────────────────────────────────────────

function generateId(): string {
  if (typeof crypto !== 'undefined' && typeof crypto.randomUUID === 'function') {
    return crypto.randomUUID();
  }
  return `msg_${Date.now()}_${Math.random().toString(36).slice(2)}`;
}

function buildAuthHeaders(authToken: string | null): Record<string, string> {
  const headers: Record<string, string> = {
    'Content-Type': 'application/json',
  };
  if (authToken !== null && authToken.length > 0) {
    headers['Authorization'] = `Bearer ${authToken}`;
  }
  return headers;
}

type ProxyErrorBody = {
  error?: string;
  message?: string;
};

async function extractErrorMessage(res: Response): Promise<string> {
  try {
    const body = (await res.json()) as ProxyErrorBody;
    return body.error ?? body.message ?? `HTTP ${res.status}`;
  } catch {
    return `HTTP ${res.status}`;
  }
}

/**
 * POST a new shift message through the secure proxy.
 * Assigns a client-generated id, createdAt timestamp, and initial status.
 * Emits analytics based on message type.
 */
export async function sendShiftMessage(
  msg: Omit<ShiftMessage, 'id' | 'createdAt' | 'status'>,
  proxyUrl: string,
  authToken: string | null,
): Promise<ShiftMessage> {
  const now = Date.now();

  const outbound: ShiftMessage = {
    ...msg,
    id: generateId(),
    createdAt: now,
    status: 'sent',
  };

  const res = await fetch(`${proxyUrl}/shift-messages`, {
    method: 'POST',
    headers: buildAuthHeaders(authToken),
    body: JSON.stringify(outbound),
  });

  if (!res.ok) {
    const detail = await extractErrorMessage(res);
    throw new Error(`sendShiftMessage failed: ${detail}`);
  }

  // The proxy may return an enriched record (e.g. server-assigned id).
  // We accept it if the shape is valid, otherwise fall back to the outbound record.
  let confirmed: ShiftMessage = outbound;
  try {
    const body = (await res.json()) as Partial<ShiftMessage>;
    if (
      typeof body.id === 'string' &&
      typeof body.shiftId === 'string' &&
      typeof body.createdAt === 'number'
    ) {
      confirmed = {
        ...outbound,
        ...body,
      } as ShiftMessage;
    }
  } catch {
    // Proxy returned no body or non-JSON — use outbound record as source of truth.
  }

  // ── Analytics ──
  if (confirmed.type === 'broadcast') {
    await trackEvent('schedule_published');
  }

  return confirmed;
}

// ─── getShiftMessages ─────────────────────────────────────────────────────────

type SupabaseRow = {
  id: string;
  shift_id: string;
  template_id?: string | null;
  date: string;
  role: string;
  type: string;
  body: string;
  sent_by: string;
  created_at: string | number;
  status: string;
  claimed_by?: string | null;
};

type SupabaseErrorBody = {
  message?: string;
  error?: string;
  hint?: string;
};

function rowToMessage(row: SupabaseRow): ShiftMessage {
  return {
    id: row.id,
    shiftId: row.shift_id,
    templateId: row.template_id ?? undefined,
    date: row.date,
    role: row.role,
    type: row.type as ShiftMessage['type'],
    body: row.body,
    sentBy: row.sent_by,
    createdAt:
      typeof row.created_at === 'number'
        ? row.created_at
        : new Date(row.created_at).getTime(),
    status: row.status as ShiftMessage['status'],
    claimedBy: row.claimed_by ?? undefined,
  };
}

/**
 * Fetch all messages for a given shift directly from Supabase REST API.
 * Results are ordered oldest-first (ascending createdAt).
 */
export async function getShiftMessages(
  shiftId: string,
  supabaseUrl: string,
  anonKey: string,
): Promise<ShiftMessage[]> {
  const url = new URL(`${supabaseUrl}/rest/v1/shift_messages`);
  url.searchParams.set('shift_id', `eq.${shiftId}`);
  url.searchParams.set('order', 'created_at.asc');

  const res = await fetch(url.toString(), {
    method: 'GET',
    headers: {
      apikey: anonKey,
      Authorization: `Bearer ${anonKey}`,
      Accept: 'application/json',
    },
  });

  if (!res.ok) {
    let detail = `HTTP ${res.status}`;
    try {
      const body = (await res.json()) as SupabaseErrorBody;
      detail = body.message ?? body.error ?? detail;
    } catch {
      // Non-JSON error body — use status code only.
    }
    throw new Error(`getShiftMessages failed: ${detail}`);
  }

  const rows = (await res.json()) as SupabaseRow[];
  return rows.map(rowToMessage);
}

// ─── claimCoverage ────────────────────────────────────────────────────────────

/**
 * Claim an open coverage/swap request via the secure proxy.
 * Emits a `shift_assigned` analytics event on success.
 */
export async function claimCoverage(
  messageId: string,
  workerId: string,
  proxyUrl: string,
  authToken: string | null,
): Promise<void> {
  const res = await fetch(`${proxyUrl}/shift-messages/${encodeURIComponent(messageId)}/claim`, {
    method: 'POST',
    headers: buildAuthHeaders(authToken),
    body: JSON.stringify({ workerId }),
  });

  if (!res.ok) {
    const detail = await extractErrorMessage(res);
    throw new Error(`claimCoverage failed: ${detail}`);
  }

  // ── Analytics ──
  await trackEvent('shift_assigned');
}

/**

  useState,

  useRef,
  useEffect,

import {

import { ThemedText } from '@/components/ThemedText';

// ─── Confirmation Toast ───────────────────────────────────────────────────────

function ConfirmationToast({ visible }: { visible: boolean }): React.ReactElement {
  const opacity = useRef(new Animated.Value(0)).current;

  useEffect(() => {
    if (visible) {
      Animated.sequence([
        Animated.timing(opacity, {
          toValue: 1,
          duration: 200,
          useNativeDriver: true,
        }),
        Animated.delay(1400),
        Animated.timing(opacity, {
          toValue: 0,
          duration: 400,
          useNativeDriver: true,
        }),
      ]).start();
    }
  }, [visible, opacity]);

  return (
    <Animated.View style={[styles.toast, { opacity }]} pointerEvents="none">
      <ThemedText style={styles.toastText}>✓ Message sent</ThemedText>
    </Animated.View>
  );
}

// ─── Template Row ─────────────────────────────────────────────────────────────

type TemplateRowProps = {
  template: MessageTemplate;
  onSelect: (template: MessageTemplate) => void;
  disabled: boolean;
};

function TemplateRow({ template, onSelect, disabled }: TemplateRowProps): React.ReactElement {
  const handlePress = useCallback(() => {
    onSelect(template);
  }, [onSelect, template]);

  return (
    <Pressable
      style={({ pressed }) => [
        styles.templateRow,
        pressed && styles.templateRowPressed,
        disabled && styles.templateRowDisabled,
      ]}
      onPress={handlePress}
      disabled={disabled}
      accessibilityRole="button"
      accessibilityLabel={template.label}
    >
      <View style={styles.templateRowInner}>
        <ThemedText style={styles.templateLabel}>{template.label}</ThemedText>
        <ThemedText style={styles.templateBody} numberOfLines={2}>
          {template.body}
        </ThemedText>
      </View>
      <ThemedText style={styles.templateChevron}>›</ThemedText>
    </Pressable>
  );
}

// ─── Main Component ───────────────────────────────────────────────────────────

export default function ShiftMessageActions({
  shiftId,
  date,
  role,
  startTime,
  endTime,
  onMessageSent,
}: Props): React.ReactElement {
  const [sheetVisible, setSheetVisible] = useState(false);
  const [sheetMode, setSheetMode] = useState<SheetMode>('template_picker');
  const [freeText, setFreeText] = useState('');
  const [sending, setSending] = useState(false);
  const [toastVisible, setToastVisible] = useState(false);
  const [errorText, setErrorText] = useState<string | null>(null);
  const toastTimer = useRef<ReturnType<typeof setTimeout> | null>(null);

  // ── Helpers ──

  const showToast = useCallback(() => {
    if (toastTimer.current !== null) {
      clearTimeout(toastTimer.current);
    }
    setToastVisible(true);
    toastTimer.current = setTimeout(() => {
      setToastVisible(false);
      toastTimer.current = null;
    }, 2000);
  }, []);

  useEffect(() => {
    return () => {
      if (toastTimer.current !== null) {
        clearTimeout(toastTimer.current);
      }
    };
  }, []);

  const buildBaseMessage = useCallback(
    (
      type: ShiftMessage['type'],
      body: string,
      templateId?: string,
    ): Omit<ShiftMessage, 'id' | 'createdAt' | 'status'> => ({
      shiftId,
      date,
      role,
      type,
      body,
      sentBy: SUPABASE_ANON_KEY.slice(0, 8), // placeholder — replace with real user id
      templateId,
    }),
    [shiftId, date, role],
  );

  const dispatchMessage = useCallback(
    async (payload: Omit<ShiftMessage, 'id' | 'createdAt' | 'status'>): Promise<void> => {
      setSending(true);
      setErrorText(null);
      try {
        const sent = await sendShiftMessage(payload, PROXY_URL, null);
        onMessageSent?.(sent);
        showToast();
        setSheetVisible(false);
        setFreeText('');
        setSheetMode('template_picker');
      } catch (err: unknown) {
        const message =
          err instanceof Error ? err.message : 'Failed to send message. Please try again.';
        setErrorText(message);
      } finally {
        setSending(false);
      }
    },
    [onMessageSent, showToast],
  );

  // ── Button handlers ──

  const handleRequestCoverage = useCallback(() => {
    setSheetMode('template_picker');
    setErrorText(null);
    setSheetVisible(true);
  }, []);

  const handleNotifyEligible = useCallback(async () => {
    const template = PRESET_TEMPLATES.find((t) => t.id === 'tpl_shift_available');
    if (template === undefined) return;
    const payload = buildBaseMessage('broadcast', template.body, template.id);
    await dispatchMessage(payload);
  }, [buildBaseMessage, dispatchMessage]);

  const handleConfirmReplacement = useCallback(() => {
    // Placeholder: hook into scheduling confirmation layer.
    // For now we send a schedule-change notice as a broadcast.
    const template = PRESET_TEMPLATES.find((t) => t.id === 'tpl_schedule_change');
    if (template === undefined) return;
    const payload = buildBaseMessage('broadcast', template.body, template.id);
    void dispatchMessage(payload);
  }, [buildBaseMessage, dispatchMessage]);

  // ── Sheet handlers ──

  const handleCloseSheet = useCallback(() => {
    if (sending) return;
    setSheetVisible(false);
    setFreeText('');
    setSheetMode('template_picker');
    setErrorText(null);
  }, [sending]);

  const handleSelectTemplate = useCallback(
    (template: MessageTemplate) => {
      const payload = buildBaseMessage(template.type, template.body, template.id);
      void dispatchMessage(payload);
    },
    [buildBaseMessage, dispatchMessage],
  );

  const handleOpenFreeText = useCallback(() => {
    setSheetMode('free_text');
    setErrorText(null);
  }, []);

  const handleBackToTemplates = useCallback(() => {
    setSheetMode('template_picker');
    setErrorText(null);
  }, []);

  const handleSendFreeText = useCallback(() => {
    const trimmed = freeText.trim();
    if (trimmed.length === 0) return;
    const payload = buildBaseMessage('coverage_request', trimmed);
    void dispatchMessage(payload);
  }, [freeText, buildBaseMessage, dispatchMessage]);

  // ─── Render ───────────────────────────────────────────────────────────────

  return (
    <View style={styles.container}>

      {/* ── Action Buttons ── */}
      <View style={styles.buttonRow}>
        <Pressable
          style={({ pressed }) => [
            styles.actionButton,
            styles.actionButtonCoverage,
            pressed && styles.actionButtonPressed,
            sending && styles.actionButtonDisabled,
          ]}
          onPress={handleRequestCoverage}
          disabled={sending}
          accessibilityRole="button"
          accessibilityLabel="Request coverage"
        >
          <ThemedText style={styles.actionButtonText}>Request coverage</ThemedText>
        </Pressable>

        <Pressable
          style={({ pressed }) => [
            styles.actionButton,
            styles.actionButtonBroadcast,
            pressed && styles.actionButtonPressed,
            sending && styles.actionButtonDisabled,
          ]}
          onPress={() => { void handleNotifyEligible(); }}
          disabled={sending}
          accessibilityRole="button"
          accessibilityLabel="Notify eligible staff"
        >
          <ThemedText style={styles.actionButtonText}>Notify eligible staff</ThemedText>
        </Pressable>

        <Pressable
          style={({ pressed }) => [
            styles.actionButton,
            styles.actionButtonConfirm,
            pressed && styles.actionButtonPressed,
            sending && styles.actionButtonDisabled,
          ]}
          onPress={handleConfirmReplacement}
          disabled={sending}
          accessibilityRole="button"
          accessibilityLabel="Confirm replacement"
        >
          <ThemedText style={styles.actionButtonText}>Confirm replacement</ThemedText>
        </Pressable>
      </View>

      {/* ── Toast ── */}
      <ConfirmationToast visible={toastVisible} />

      {/* ── Bottom Sheet Modal ── */}
      <Modal
        visible={sheetVisible}
        animationType="slide"
        transparent
        onRequestClose={handleCloseSheet}
        statusBarTranslucent
      >
        <Pressable style={styles.backdrop} onPress={handleCloseSheet} />

        <KeyboardAvoidingView
          behavior={Platform.OS === 'ios' ? 'padding' : 'height'}
          style={styles.sheetWrapper}
        >
          <View style={styles.sheet}>

            {/* Sheet Header */}
            <View style={styles.sheetHeader}>
              {sheetMode === 'free_text' ? (
                <Pressable
                  onPress={handleBackToTemplates}
                  accessibilityRole="button"
                  accessibilityLabel="Back to templates"
                >
                  <ThemedText style={styles.sheetBackText}>‹ Templates</ThemedText>
                </Pressable>
              ) : (
                <View />
              )}
              <ThemedText style={styles.sheetTitle}>
                {sheetMode === 'template_picker' ? 'Send message' : 'Custom message'}
              </ThemedText>
              <Pressable
                onPress={handleCloseSheet}
                disabled={sending}
                accessibilityRole="button"
                accessibilityLabel="Close"
              >
                <ThemedText style={styles.sheetCloseText}>✕</ThemedText>
              </Pressable>
            </View>

            {/* Shift context summary */}
            <View style={styles.contextBadge}>
              <ThemedText style={styles.contextText}>
                {role} · {date} · {startTime}–{endTime}
              </ThemedText>
            </View>

            {/* Error */}
            {errorText !== null && (
              <View style={styles.errorBanner}>
                <ThemedText style={styles.errorText}>{errorText}</ThemedText>
              </View>
            )}

            {/* ── Template Picker ── */}
            {sheetMode === 'template_picker' && (
              <ScrollView
                style={styles.templateList}
                contentContainerStyle={styles.templateListContent}
                keyboardShouldPersistTaps="handled"
              >
                {PRESET_TEMPLATES.map((template) => (
                  <TemplateRow
                    key={template.id}
                    template={template}
                    onSelect={handleSelectTemplate}
                    disabled={sending}
                  />
                ))}

                {/* Free-text option */}
                <Pressable
                  style={({ pressed }) => [
                    styles.templateRow,
                    styles.freeTextRow,
                    pressed && styles.templateRowPressed,
                    sending && styles.templateRowDisabled,
                  ]}
                  onPress={handleOpenFreeText}
                  disabled={sending}
                  accessibilityRole="button"
                  accessibilityLabel="Write a custom message"
                >
                  <View style={styles.templateRowInner}>
                    <ThemedText style={styles.templateLabel}>Write custom message…</ThemedText>
                    <ThemedText style={styles.templateBody}>
                      Compose your own message for this shift.
                    </ThemedText>
                  </View>
                  <ThemedText style={styles.templateChevron}>›</ThemedText>
                </Pressable>
              </ScrollView>
            )}

            {/* ── Free Text Composer ── */}
            {sheetMode === 'free_text' && (
              <View style={styles.freeTextComposer}>
                <TextInput
                  style={styles.freeTextInput}
                  multiline
                  placeholder="Type your message…"
                  placeholderTextColor="#9CA3AF"
                  value={freeText}
                  onChangeText={setFreeText}
                  editable={!sending}
                  autoFocus
                  returnKeyType="default"
                  maxLength={500}
                  accessibilityLabel="Custom message text"
                />

                    styles.sendButton,

            {/* Sending indicator (template flow) */}
            {sending && sheetMode === 'template_picker' && (

                <ActivityIndicator color="#0D9488" size="large" />
                <ThemedText style={styles.sendingText}>Sending…</ThemedText>
              </View>
            )}
          </View>
        </KeyboardAvoidingView>
      </Modal>
    </View>
  );
}
