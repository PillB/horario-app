// components/CapabilityMatrix.tsx
/**
 * CapabilityMatrix — displays a feature comparison table (Free vs Pro).
 *
 * - 3-column layout: Feature | Free | Pro
 * - Highlights rows listed in `highlightMissing` with amber background
 * - Shows an "Upgrade to Pro" CTA when any pro-only row is highlighted
 * - Fully accessible and i18n-ready
 */
import React from 'react';
import {
  View,
  Pressable,
  ScrollView,
  StyleSheet,
  AccessibilityInfo,
} from 'react-native';
import { ThemedText } from '@/components/ThemedText';
import { ThemedView } from '@/components/ThemedView';
import { brand, Colors, ThemeColors } from '@/constants/Colors';
import { useColorScheme } from '@/hooks/useColorScheme';
import { useT } from '@/contexts/LocaleContext';

// ─── Types ────────────────────────────────────────────────────────────────────

type Tier = 'free' | 'pro';

type FeatureRow = {
  /** Stable key used for i18n lookup and highlighting */
  key: string;
  /** i18n key, e.g. "capability.autoAssign" */
  i18nKey: string;
  /** English fallback rendered when translation is missing */
  fallback: string;
  /** Which tier includes this feature */
  tier: Tier;
};

export type CapabilityMatrixProps = {
  /** Called when the user taps the "Upgrade to Pro" CTA */
  onUpgradePress?: () => void;
  /**
   * Feature keys (matching FeatureRow.key) that should be highlighted in amber,
   * typically because the current user is blocked from using them.
   */
  highlightMissing?: string[];
};

// ─── Sub-components ───────────────────────────────────────────────────────────

function CheckIcon({ color }: { color: string }) {
  // Unicode heavy check mark — universally supported, no image dependency
  return (
    <ThemedText
      style={{ color, fontSize: 16, fontWeight: '700' }}
      accessibilityLabel="included">
      {'✓'}
    </ThemedText>
  );
}

function LockIcon({ color }: { color: string }) {
  return (
    <ThemedText
      style={{ color, fontSize: 14 }}
      accessibilityLabel="pro only, locked">
      {'🔒'}
    </ThemedText>
  );
}

function DashIcon({ color }: { color: string }) {
  return (
    <ThemedText
      style={{ color, fontSize: 16, fontWeight: '300' }}
      accessibilityLabel="not included">
      {'–'}
    </ThemedText>
  );
}

// ─── Main component ───────────────────────────────────────────────────────────

export function CapabilityMatrix({
  onUpgradePress,
  highlightMissing = [],
}: CapabilityMatrixProps) {
  const colorScheme = useColorScheme();
  const colors = Colors[colorScheme ?? 'light'];
  const styles = makeStyles(colors);
  const { t } = useT();

  const highlightSet = new Set(highlightMissing);
  const hasHighlightedPro = FEATURE_ROWS.some(
    (row) => row.tier === 'pro' && highlightSet.has(row.key),
  );

  const resolveLabel = (row: FeatureRow): string => {
    try {
      const translated = t(row.i18nKey as Parameters<typeof t>[0]);
      // If the translation key is returned as-is, fall back to English
      return translated && translated !== row.i18nKey ? translated : row.fallback;
    } catch {
      return row.fallback;
    }
  };

  const resolveString = (key: string, fallback: string): string => {
    try {
      const translated = t(key as Parameters<typeof t>[0]);
      return translated && translated !== key ? translated : fallback;
    } catch {
      return fallback;
    }
  };

  return (
    <ThemedView style={styles.container}>
      {/* Table header */}
      <View
        style={styles.headerRow}
        accessibilityRole="header"
        accessible>
        <View style={[styles.cell, styles.featureCell]}>
          <ThemedText style={styles.headerText}>
            {resolveString('capability.header.feature', 'Feature')}
          </ThemedText>
        </View>
        <View style={[styles.cell, styles.tierCell]}>
          <ThemedText style={styles.headerText}>
            {resolveString('capability.header.free', 'Free')}
          </ThemedText>
        </View>
        <View style={[styles.cell, styles.tierCell]}>
          <View style={styles.proHeaderContent}>
            <ThemedText style={[styles.headerText, styles.proHeaderText]}>
              {resolveString('capability.header.pro', 'Pro')}
            </ThemedText>
            <View style={styles.proBadge}>
              <ThemedText style={styles.proBadgeText}>
                {resolveString('capability.header.proBadge', '★')}
              </ThemedText>
            </View>
          </View>
        </View>
      </View>

      <View style={styles.divider} accessibilityElementsHidden />

      {/* Table rows */}
      <ScrollView
        scrollEnabled={false}
        accessible={false}>
        {FEATURE_ROWS.map((row, index) => {
          const isHighlighted = highlightSet.has(row.key);
          const isPro = row.tier === 'pro';
          const isLastRow = index === FEATURE_ROWS.length - 1;
          const label = resolveLabel(row);

          const rowA11yLabel = isPro
            ? `${label}, Pro only, ${isHighlighted ? 'currently locked' : 'available with Pro'}`
            : `${label}, included in Free and Pro`;

          return (
            <View key={row.key}>
              <View
                style={[
                  styles.row,
                  index % 2 === 0 ? styles.rowEven : styles.rowOdd,
                  isHighlighted && styles.rowHighlighted,
                  isLastRow && styles.rowLast,
                ]}
                accessible
                accessibilityLabel={rowA11yLabel}>
                {/* Feature label */}
                <View style={[styles.cell, styles.featureCell]}>
                  <View style={styles.featureLabelRow}>
                    {isHighlighted && (
                      <View
                        style={styles.highlightDot}
                        accessibilityElementsHidden
                      />
                    )}
                    <ThemedText
                      style={[
                        styles.featureText,
                        isHighlighted && styles.featureTextHighlighted,
                      ]}
                      numberOfLines={2}>
                      {label}
                    </ThemedText>
                  </View>
                </View>

                {/* Free column */}
                <View
                  style={[styles.cell, styles.tierCell, styles.centeredCell]}
                  accessibilityElementsHidden>
                  {isPro ? (
                    <DashIcon color={colors.textSecondary} />
                  ) : (
                    <CheckIcon color={brand.primary} />
                  )}
                </View>

                {/* Pro column */}
                <View
                  style={[styles.cell, styles.tierCell, styles.centeredCell]}
                  accessibilityElementsHidden>
                  {isPro ? (
                    isHighlighted ? (
                      <LockIcon color={brand.accentDark} />
                    ) : (
                      <CheckIcon color={brand.pro} />
                    )
                  ) : (
                    <CheckIcon color={brand.pro} />
                  )}
                </View>
              </View>

              {!isLastRow && (
                <View style={styles.rowDivider} accessibilityElementsHidden />
              )}
            </View>
          );
        })}
      </ScrollView>

      {/* Upgrade CTA */}
      {hasHighlightedPro && onUpgradePress && (
        <>
          <View style={styles.divider} accessibilityElementsHidden />
          <View style={styles.ctaContainer}>
            <ThemedText style={styles.ctaHint}>
              {resolveString(
                'capability.cta.hint',
                'Unlock Pro features to remove these restrictions.',
              )}
            </ThemedText>
            <Pressable
              onPress={onUpgradePress}
              style={({ pressed }) => [
                styles.ctaButton,
                pressed && styles.ctaButtonPressed,
              ]}
              accessibilityRole="button"
              accessibilityLabel={resolveString(
                'capability.cta.a11yLabel',
                'Upgrade to Pro to unlock all features',
              )}
              accessibilityHint={resolveString(
                'capability.cta.a11yHint',
                'Opens the billing page to start your Pro subscription',
              )}>
              <ThemedText style={styles.ctaButtonText}>
                {resolveString('capability.cta.button', '🚀 Upgrade to Pro')}
              </ThemedText>
            </Pressable>
            <ThemedText style={styles.ctaPricingNote}>
              {resolveString(
                'capability.cta.pricingNote',
                'Plans start at $9 / month · Cancel anytime',
              )}
            </ThemedText>
          </View>
        </>
      )}
    </ThemedView>
  );
}
