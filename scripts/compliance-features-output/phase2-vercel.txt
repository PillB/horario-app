// NOTE: This route explicitly uses the Node.js runtime because it depends on
// the built-in `crypto` module (createPublicKey / createVerify) for RSA-SHA256
// JWT verification. The Web Crypto API available in Edge runtimes does not
// expose the synchronous RSA verification path required by Clerk's JWKS flow.
// If Clerk ever ships an Edge-compatible verification SDK this can be switched
// to `export const runtime = 'edge'` for lower cold-start latency.
//
// TODO: Implement per-IP or per-account rate limiting before going to
// production at scale. Options:
//   • Upstash Redis + @upstash/ratelimit (sliding window, 60 req / 60 s)
//   • Vercel KV with a simple token-bucket stored under `rl:<ip>`
//   • An edge middleware (middleware.ts) that short-circuits before this
//     function is invoked, keeping this function's cold-start budget clean.
// Until that is in place, Supabase RLS and the Clerk JWT check provide the
// primary defence against abuse.

// Maximum age of a valid JWT in seconds beyond which we reject it regardless
// of the `exp` claim.  Acts as a defence-in-depth guard.

const IS_PRODUCTION = process.env.NODE_ENV === 'production';

// ─── JWKS / JWT helpers ───────────────────────────────────────────────────────

type JWK = { kid?: string; kty: string; n: string; e: string; alg?: string; use?: string };
type JWKSet = { keys: JWK[] };

let jwksCache: JWKSet | null = null;
let jwksCacheExpiry = 0;

async function fetchJwks(): Promise<JWKSet> {
  if (jwksCache && Date.now() < jwksCacheExpiry) return jwksCache;
  if (!CLERK_JWKS_URL) throw new Error('CLERK_JWKS_URL is not configured');
  const res = await fetch(CLERK_JWKS_URL, {
    // Explicit timeout via AbortSignal — avoids hanging the Vercel function.
    signal: AbortSignal.timeout(5000),
  });
  if (!res.ok) throw new Error(`Failed to fetch JWKS (status ${res.status})`);
  jwksCache = (await res.json()) as JWKSet;
  jwksCacheExpiry = Date.now() + 5 * 60 * 1000;
  return jwksCache;
}

async function verifyClerkToken(token: string): Promise<string | null> {
  try {
    const parts = token.split('.');
    if (parts.length !== 3) return null;

    const header = JSON.parse(
      Buffer.from(parts[0], 'base64url').toString()
    ) as { kid?: string; alg?: string };

    const payload = JSON.parse(
      Buffer.from(parts[1], 'base64url').toString()
    ) as { sub?: string; exp?: number; iat?: number };

    const now = Date.now() / 1000;

    // Reject expired tokens.
    if (payload.exp && now > payload.exp) return null;

    // Reject tokens issued too far in the past (defence-in-depth).
    if (payload.iat && now - payload.iat > MAX_TOKEN_AGE_S) return null;

    const jwks = await fetchJwks();
    const jwk = header.kid
      ? jwks.keys.find((k) => k.kid === header.kid)
      : jwks.keys[0];
    if (!jwk) return null;

    const publicKey = createPublicKey({
      key: jwk as unknown as CryptoJWK,
      format: 'jwk',
    });
    const verifier = createVerify('RSA-SHA256');
    verifier.update(`${parts[0]}.${parts[1]}`);
    const valid = verifier.verify(publicKey, parts[2], 'base64url');
    return valid ? (payload.sub ?? null) : null;
  } catch {
    // Swallow verification errors — return null to indicate invalid token.
    // Do NOT log the raw error here; it may contain token fragments.
    return null;
  }
}

async function getAuthorizedAccountId(request: Request): Promise<string | null> {
  if (!AUTH_ENABLED) return null;
  const auth = request.headers.get('Authorization');
  if (!auth?.startsWith('Bearer ')) return null;
  return verifyClerkToken(auth.slice(7));
}

// ─── Request size guard ───────────────────────────────────────────────────────

// Reads the raw body as an ArrayBuffer so we can check its byte length before
// handing it to JSON.parse.  Returns null if the body exceeds MAX_BODY_BYTES.
async function readBodySafe(
  request: Request
): Promise<{ text: string } | { error: string; status: number }> {
  // Honour Content-Length if present for a cheap early-rejection path.
  const contentLength = request.headers.get('content-length');
  if (contentLength !== null) {
    const declared = parseInt(contentLength, 10);
    if (!Number.isNaN(declared) && declared > MAX_BODY_BYTES) {
      return { error: 'Request body too large (max 1 MiB)', status: 413 };
    }
  }

  let buffer: ArrayBuffer;
  try {
    buffer = await request.arrayBuffer();
  } catch {
    return { error: 'Failed to read request body', status: 400 };
  }

  if (buffer.byteLength > MAX_BODY_BYTES) {
    return { error: 'Request body too large (max 1 MiB)', status: 413 };
  }

  return { text: new TextDecoder().decode(buffer) };
}

const fetchHistoricalShifts = async (
  client: SupabaseClient,
  filters?: ForecastFilters
): Promise<HistoricalShift[]> => {
  const query = client
    .from('shifts')
    .select('date, role, worker_id, hours')
    .order('date', { ascending: true })
    .limit(500);

  if (filters?.date) {
    query.gte('date', filters.date).lte('date', filters.date);

  if (filters?.role) {
    query.eq('role', filters.role);

  const { data, error } = await query;
  if (error) {
    console.warn('[secure-proxy] Unable to fetch historical shifts', error.message);
    return [];

  return (data ?? []).map((shift) => ({
    date: shift.date,
    role: shift.role,
    hours: shift.hours ?? 8,
    worker_id: shift.worker_id,
  }));

const persistForecasts = async (

// ─── External forecast API call ───────────────────────────────────────────────

type ForecastApiOutcome = {
  forecasts: DemandForecast[];
  attempted: boolean;
  failed: boolean;
  failureReason?: string;
};

const callForecastApi = async (inputs: {
  shifts: HistoricalShift[];
  signals: ExternalSignal[];
  filters?: ForecastFilters;
}): Promise<ForecastApiOutcome> => {
  if (!FORECAST_API_URL) {
    return { forecasts: [], attempted: false, failed: false };
  }

  try {
    const response = await fetch(FORECAST_API_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(inputs),
      signal: AbortSignal.timeout(15_000),
    });

    if (!response.ok) {
      const failureReason = `Forecast API responded with status ${response.status}`;
      console.warn('[secure-proxy]', failureReason);
      return { forecasts: [], attempted: true, failed: true, failureReason };
    }

    const apiPayload = (await response.json()) as DemandForecast[];
    const normalized = apiPayload.map((item) => ({
      ...item,
      required: Number(item.required ?? 0),
      source: item.source ?? 'model',
    }));

    return { forecasts: normalized, attempted: true, failed: false };
  } catch (error) {
    const failureReason =
      error instanceof Error ? error.message : 'Unknown forecast API error';
    console.warn('[secure-proxy] Forecast API call failed', failureReason);
    return { forecasts: [], attempted: true, failed: true, failureReason };
  }
};

const computeDailyAverages = (
  shifts: HistoricalShift[],
  signals: ExternalSignal[],
  filters?: ForecastFilters
): DemandForecast[] => {
  const grouped = shifts.reduce<Record<string, number[]>>((acc, shift) => {
    if (filters?.role && shift.role !== filters.role) return acc;
    const key = buildKey({ date: shift.date, role: shift.role });
    if (!acc[key]) acc[key] = [];
    acc[key].push(shift.hours ?? 8);
    return acc;
  }, {});

  const signalMap = signals.reduce<Record<string, number>>((acc, signal) => {
    if (filters?.role && signal.role && signal.role !== filters.role) return acc;
    if (!signal.weight) return acc;
    const key = buildKey({ date: signal.date, role: signal.role ?? 'global' });
    acc[key] = (acc[key] ?? 0) + signal.weight;
    return acc;
  }, {});

  const keys = new Set<string>();
  Object.keys(grouped).forEach((k) => keys.add(k));
  Object.keys(signalMap).forEach((k) => keys.add(k));
  if (filters?.date && filters?.role) {
    keys.add(buildKey({ date: filters.date, role: filters.role }));
  }

  const forecasts: DemandForecast[] = [];

  keys.forEach
